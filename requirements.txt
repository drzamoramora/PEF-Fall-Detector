# Exact versions of the reference environment (the Mac PEF-FallDB was measured
# on; phase 8.4: 110/128 there, 115/128 on cloud x86). Pinned with == and not
# >= because MediaPipe's landmarks change with the OpenCV and numpy versions
# (clips A01-S4, A14-S1, A17-S4: same code and config, another verdict with
# OpenCV 5.0).
#
# The direct dependencies and their groups are declared in pyproject.toml.
# This file is the lock: to be regenerated on the reference Mac with
#   pip freeze --exclude-editable > requirements.txt
# (B9; README, "Install"). The Raspberry Pi gets its own lock, frozen there.
#
# Install in a clean venv:
#   python3.11 -m venv venv && source venv/bin/activate
#   pip install -r requirements.txt && pip install -e . --no-deps

mediapipe==0.10.14

# different versions, the last one installed wins.
opencv-contrib-python==4.13.0.92
